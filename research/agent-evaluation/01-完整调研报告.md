## 2. 评估指标体系

### 2.1 核心指标定义

```python
from dataclasses import dataclass
from typing import List, Dict, Optional
from enum import Enum

class TaskStatus(Enum):
    SUCCESS = "success"
    PARTIAL = "partial"
    FAILED = "failed"

@dataclass
class AgentMetrics:
    """Agent 评估指标"""

    # 任务完成指标
    task_success_rate: float  # 任务成功率
    task_completion_quality: float  # 完成质量 (0-1)
    average_steps_to_complete: float  # 平均完成步数

    # 准确性指标
    factual_accuracy: float  # 事实准确性
    logical_consistency: float  # 逻辑一致性
    hallucination_rate: float  # 幻觉率

    # 工具使用指标
    tool_selection_accuracy: float  # 工具选择准确率
    parameter_correctness: float  # 参数正确率
    tool_error_recovery: float  # 工具错误恢复率

    # 效率指标
    average_response_time: float  # 平均响应时间 (ms)
    token_efficiency: float  # Token 使用效率

    # 安全指标
    safety_violation_rate: float  # 安全违规率
    privacy_leak_rate: float  # 隐私泄露率

    def overall_score(self) -> float:
        """计算综合分数"""
        weights = {
            'task_success_rate': 0.25,
            'task_completion_quality': 0.20,
            'factual_accuracy': 0.20,
            'tool_selection_accuracy': 0.15,
            'average_response_time': 0.10,
            'safety_violation_rate': 0.10,
        }

        # 反向指标处理
        safety_score = 1 - self.safety_violation_rate
        time_score = max(0, 1 - self.average_response_time / 10000)

        score = (
            self.task_success_rate * weights['task_success_rate'] +
            self.task_completion_quality * weights['task_completion_quality'] +
            self.factual_accuracy * weights['factual_accuracy'] +
            self.tool_selection_accuracy * weights['tool_selection_accuracy'] +
            time_score * weights['average_response_time'] +
            safety_score * weights['safety_violation_rate']
        )

        return score
```

### 2.2 指标计算公式

```python
class MetricCalculator:
    """指标计算器"""

    @staticmethod
    def success_rate(results: List[TaskResult]) -> float:
        """计算成功率"""
        success_count = sum(1 for r in results if r.status == TaskStatus.SUCCESS)
        return success_count / len(results)

    @staticmethod
    def step_efficiency(results: List[TaskResult]) -> float:
        """计算步数效率"""
        if not results:
            return 0.0

        optimal_steps = sum(r.optimal_steps for r in results)
        actual_steps = sum(r.actual_steps for r in results)

        if actual_steps == 0:
            return 0.0

        return min(1.0, optimal_steps / actual_steps)

    @staticmethod
    def tool_accuracy(results: List[TaskResult]) -> float:
        """计算工具使用准确率"""
        total_calls = sum(r.tool_calls for r in results)
        correct_calls = sum(r.correct_tool_calls for r in results)

        if total_calls == 0:
            return 1.0

        return correct_calls / total_calls

    @staticmethod
    def hallucination_rate(claims: List[str], context: str) -> float:
        """计算幻觉率"""
        if not claims:
            return 0.0

        hallucinated = sum(1 for claim in claims if not is_supported(claim, context))
        return hallucinated / len(claims)

    @staticmethod
    def token_efficiency(response: str, information_content: float) -> float:
        """计算 Token 效率"""
        token_count = len(response.split()) * 1.3  # 估算 token 数

        if token_count == 0:
            return 0.0

        return information_content / token_count
```

```python
class LLMJudge:
    """LLM 评判者"""

    def __init__(self, llm):
        self.llm = llm

    def evaluate_task(self, task: str, response: str,
                      criteria: List[str]) -> EvaluationResult:

        prompt = f"""Evaluate the following response based on the task and criteria.

Task: {task}
Response: {response}

Criteria:
{self._format_criteria(criteria)}

Rate each criterion from 1-5 and provide brief justification.
Format: JSON with scores and comments."""

        response = self.llm.generate(prompt)
        return self._parse_evaluation(response)

    def compare_responses(self, task: str, response_a: str,
                          response_b: str) -> str:

        prompt = f"""Compare two responses to the same task.

Task: {task}

Response A:
{response_a}

Response B:
{response_b}

Which response is better? Consider: accuracy, completeness, helpfulness.
Answer: A or B, with explanation."""

        return self.llm.generate(prompt)

    def _format_criteria(self, criteria: List[str]) -> str:
        return "\n".join(f"- {c}" for c in criteria)
```

### 4.2 基于规则的评估

```python
class RuleBasedEvaluator:
    """基于规则的评估器"""

    def __init__(self, rules: List[Dict]):
        self.rules = rules

    def evaluate(self, response: str, context: dict) -> EvaluationReport:
        results = {}

        for rule in self.rules:
            rule_name = rule['name']
            condition = rule['condition']

            # 执行规则检查
            passed = self._check_condition(condition, response, context)
            results[rule_name] = {
                'passed': passed,
                'weight': rule.get('weight', 1.0)
            }

        # 计算总分
        total_score = sum(
            r['weight'] * (1 if r['passed'] else 0)
            for r in results.values()
        )
        max_score = sum(r['weight'] for r in results.values())

        return EvaluationReport(
            score=total_score / max_score,
            details=results
        )

    def _check_condition(self, condition: str,
                         response: str, context: dict) -> bool:
        # 执行条件检查 (可以是正则、关键字、逻辑判断等)
        import re
        if condition.startswith('regex:'):
            pattern = condition[6:]
            return bool(re.search(pattern, response))
        elif condition.startswith('contains:'):
            keyword = condition[9:]
            return keyword in response
        elif condition.startswith('length_min:'):
            min_len = int(condition[11:])
            return len(response) >= min_len
        return True
```

### 4.3 基于执行的评估

```python
class ExecutionBasedEvaluator:
    """基于执行的评估器 (适用于代码/工具调用)"""

    def evaluate_code(self, task: str, generated_code: str,
                      test_cases: List[Dict]) -> ExecutionReport:

        results = []
        for test in test_cases:
            result = self._run_test(generated_code, test)
            results.append(result)

        passed = sum(1 for r in results if r['passed'])
        total = len(results)

        return ExecutionReport(
            pass_rate=passed / total,
            test_results=results
        )

    def _run_test(self, code: str, test: Dict) -> Dict:
        # 执行测试用例
        try:
            # 运行代码并检查结果
            output = self._execute(code, test['input'])
            passed = output == test['expected_output']
            return {'passed': passed, 'output': output}
        except Exception as e:
            return {'passed': False, 'error': str(e)}

    def _execute(self, code: str, input_data: Any) -> Any:
        # 在沙箱中执行代码
        import subprocess
        result = subprocess.run(
            ['python', '-c', code],
            input=str(input_data),
            capture_output=True,
            text=True,
            timeout=10
        )
        return result.stdout
```

### 4.4 综合评估框架

```python
class ComprehensiveEvaluator:
    """综合评估框架"""

    def __init__(self, llm_judge, rule_evaluator, execution_evaluator):
        self.llm_judge = llm_judge
        self.rule_evaluator = rule_evaluator
        self.execution_evaluator = execution_evaluator

    def evaluate(self, task: EvaluationTask,
                 agent_response: AgentResponse) -> FullEvaluationReport:

        # 多维度评估
        llm_score = self.llm_judge.evaluate_task(
            task.description,
            agent_response.content,
            task.success_criteria
        )

        rule_score = self.rule_evaluator.evaluate(
            agent_response.content,
            agent_response.context
        )

        exec_score = None
        if agent_response.code_or_tool_calls:
            exec_score = self.execution_evaluator.evaluate_code(
                task.description,
                agent_response.code,
                task.test_cases
            )

        # 综合分数
        final_score = self._combine_scores(
            llm_score, rule_score, exec_score
        )

        return FullEvaluationReport(
            task_id=task.id,
            llm_score=llm_score,
            rule_score=rule_score,
            execution_score=exec_score,
            final_score=final_score
        )

    def _combine_scores(self, llm: float, rule: float,
                        exec: Optional[float]) -> float:
        if exec is not None:
            return 0.4 * llm + 0.3 * rule + 0.3 * exec
        return 0.6 * llm + 0.4 * rule
```

### 5.4 评估者间一致性

```python
from scipy import stats

class InterRaterReliability:
    """评估者间一致性分析"""

    @staticmethod
    def cohens_kappa(ratings_a: List[int], ratings_b: List[int]) -> float:
        """计算 Cohen's Kappa 系数"""
        kappa, _ = stats.cohen_kappa_score(ratings_a, ratings_b)
        return kappa

    @staticmethod
    def interpret_kappa(kappa: float) -> str:
        """解释 Kappa 系数"""
        if kappa < 0:
            return "无一致性"
        elif kappa < 0.2:
            return "轻微一致性"
        elif kappa < 0.4:
            return "一般一致性"
        elif kappa < 0.6:
            return "中等一致性"
        elif kappa < 0.8:
            return "较强一致性"
        else:
            return "几乎完全一致"
```

### 6.5 自定义评估管道

```python
class CustomEvaluationPipeline:
    """自定义评估管道"""

    def __init__(self, evaluators: List[BaseEvaluator]):
        self.evaluators = evaluators
        self.results_store = []

    def run(self, agent, test_cases: List[TestCase]) -> EvaluationReport:
        all_results = []

        for test_case in test_cases:
            # 执行 Agent
            response = agent.run(test_case.input)

            # 多维度评估
            result = TestCaseResult(
                test_case=test_case,
                response=response,
                scores={}
            )

            for evaluator in self.evaluators:
                score = evaluator.evaluate(test_case, response)
                result.scores[evaluator.name] = score

            all_results.append(result)

        # 汇总报告
        return self._generate_report(all_results)

    def _generate_report(self, results: List[TestCaseResult]) -> EvaluationReport:
        report = EvaluationReport()

        for evaluator_name in self.evaluators:
            scores = [r.scores[evaluator_name] for r in results]
            report.metrics[evaluator_name] = {
                'mean': np.mean(scores),
                'std': np.std(scores),
                'min': np.min(scores),
                'max': np.max(scores)
            }

        return report
```

```python
class MonitoringAlertSystem:
    """监控告警系统"""

    def __init__(self, config: AlertConfig):
        self.config = config
        self.alert_history = []

    def check_and_alert(self, metrics: MonitoringMetrics):
        alerts = []

        # 延迟告警
        if metrics.p95_latency > self.config.p95_latency_threshold:
            alerts.append(Alert(
                type="high_latency",
                severity="warning",
                message=f"P95 latency {metrics.p95_latency}ms exceeds threshold"
            ))

        # 错误率告警
        if metrics.error_rate > self.config.error_rate_threshold:
            alerts.append(Alert(
                type="high_error_rate",
                severity="critical",
                message=f"Error rate {metrics.error_rate} exceeds threshold"
            ))

        # 质量下降告警
        if metrics.average_user_rating < self.config.min_rating:
            alerts.append(Alert(
                type="quality_degradation",
                severity="warning",
                message=f"User rating {metrics.average_user_rating} below threshold"
            ))

        # 发送告警
        for alert in alerts:
            self._send_alert(alert)

        return alerts

    def _send_alert(self, alert: Alert):
        # 发送告警通知
        self.alert_history.append(alert)
```

### 7.3 A/B 测试框架

```python
class ABTestFramework:
    """A/B 测试框架"""

    def __init__(self):
        self.variants = {}
        self.results = {}

    def add_variant(self, name: str, agent_config: dict):
        self.variants[name] = {
            'config': agent_config,
            'traffic_percentage': 0
        }

    def set_traffic(self, variant_a: float, variant_b: float):
        self.variants['A']['traffic_percentage'] = variant_a
        self.variants['B']['traffic_percentage'] = variant_b

    def get_variant(self, user_id: str) -> str:
        # 基于用户 ID 分配变体
        hash_value = hash(user_id) % 100
        cumulative = 0
        for name, variant in self.variants.items():
            cumulative += variant['traffic_percentage']
            if hash_value < cumulative:
                return name
        return 'A'

    def record_result(self, variant: str, metric: str, value: float):
        if variant not in self.results:
            self.results[variant] = []
        self.results[variant].append({metric: value})

    def analyze(self) -> ABTestReport:
        # 统计分析
        report = ABTestReport()
        for metric in ['success_rate', 'latency', 'user_rating']:
            values_a = [r[metric] for r in self.results.get('A', [])]
            values_b = [r[metric] for r in self.results.get('B', [])]

            # t 检验
            t_stat, p_value = stats.ttest_ind(values_a, values_b)

            report.metrics[metric] = {
                'mean_a': np.mean(values_a),
                'mean_b': np.mean(values_b),
                'p_value': p_value,
                'significant': p_value < 0.05
            }

        return report
```
