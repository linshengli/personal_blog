## 1. 概念剖析

### 1.1 什么是 RAG

**RAG (Retrieval Augmented Generation，检索增强生成)** 是一种将信息检索与文本生成相结合的技术架构。它通过先从知识库中检索相关信息，再基于检索结果生成回答，有效解决了大语言模型的知识时效性和幻觉问题。

### 1.2 RAG 的核心价值

```
┌─────────────────────────────────────────────────────────────┐
│                    Why RAG?                                  │
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  Problem                          Solution                   │
│  ─────────                      ──────────                  │
│  • 知识截止 (Knowledge Cutoff)  → 实时检索最新信息          │
│  • 幻觉问题 (Hallucination)     → 基于事实生成              │
│  • 缺乏可解释性                 → 提供引用来源              │
│  • 领域知识不足                 → 接入专业知识库            │
│  • 数据隐私顾虑                 → 本地知识库部署            │
│                                                              │
└─────────────────────────────────────────────────────────────┘
```

### 1.4 RAG 工作流程

```python
class RAGPipeline:
    """RAG 管道基础实现"""

    def __init__(self, retriever, generator, embedder):
        self.retriever = retriever      # 检索器
        self.generator = generator      # 生成器 (LLM)
        self.embedder = embedder        # 嵌入模型

    def query(self, question: str, top_k: int = 5) -> str:
        """执行 RAG 查询"""

        # Step 1: 查询嵌入
        query_embedding = self.embedder.encode(question)

        # Step 2: 向量检索
        documents = self.retriever.search(
            query_embedding,
            top_k=top_k
        )

        # Step 3: 构建上下文
        context = self._build_context(documents)

        # Step 4: 生成回答
        response = self.generator.generate(
            question=question,
            context=context
        )

        # Step 5: 添加引用
        response_with_citations = self._add_citations(
            response, documents
        )

        return response_with_citations

    def _build_context(self, documents: List[Document]) -> str:
        """构建检索结果的上下文"""
        return "\n\n".join([
            f"[Document {i+1}]\n{doc.content}"
            for i, doc in enumerate(documents)
        ])
```

### 2.4 检索算法

#### 2.4.1 向量检索 (Dense Retrieval)

```python
import faiss
import numpy as np

class DenseRetriever:
    """稠密向量检索器"""

    def __init__(self, dimension: int = 1024):
        self.dimension = dimension
        self.index = None
        self.documents = []

    def build_index(self, embeddings: np.ndarray, documents: List[str]):
        """构建 HNSW 索引"""
        self.documents = documents

        # HNSW 索引
        self.index = faiss.IndexHNSWFlat(
            dimension,
            32,  # M 参数
            faiss.METRIC_INNER_PRODUCT
        )
        self.index.add(embeddings)

    def search(self, query_embedding: np.ndarray, top_k: int = 5):
        """执行检索"""
        distances, indices = self.index.search(
            query_embedding.reshape(1, -1),
            top_k
        )

        return [
            {"document": self.documents[i], "score": float(d)}
            for i, d in zip(indices[0], distances[0])
            if i != -1
        ]
```

#### 2.4.2 稀疏检索 (Sparse Retrieval)

```python
from rank_bm25 import BM25Okapi

class SparseRetriever:
    """BM25 稀疏检索器"""

    def __init__(self, documents: List[str]):
        # 分词
        tokenized_docs = [doc.lower().split() for doc in documents]
        self.bm25 = BM25Okapi(tokenized_docs)
        self.documents = documents

    def search(self, query: str, top_k: int = 5):
        # 查询分词
        query_tokens = query.lower().split()

        # 获取分数
        scores = self.bm25.get_scores(query_tokens)

        # 返回 top-k
        top_indices = np.argsort(scores)[-top_k:][::-1]

        return [
            {"document": self.documents[i], "score": float(scores[i])}
            for i in top_indices
        ]
```

#### 2.4.3 混合检索 (Hybrid Retrieval)

```python
class HybridRetriever:
    """混合检索器 - 结合稠密和稀疏检索"""

    def __init__(self, dense_retriever, sparse_retriever, alpha: float = 0.5):
        self.dense = dense_retriever
        self.sparse = sparse_retriever
        self.alpha = alpha  # 稠密检索权重

    def search(self, query: str, query_embedding: np.ndarray, top_k: int = 5):
        # 获取两种检索结果
        dense_results = self.dense.search(query_embedding, top_k=top_k * 2)
        sparse_results = self.sparse.search(query, top_k=top_k * 2)

        # 合并结果
        all_docs = {}

        for r in dense_results:
            doc_id = hash(r['document'])
            all_docs[doc_id] = {
                'document': r['document'],
                'dense_score': r['score'],
                'sparse_score': 0.0
            }

        for r in sparse_results:
            doc_id = hash(r['document'])
            if doc_id in all_docs:
                all_docs[doc_id]['sparse_score'] = r['score']
            else:
                all_docs[doc_id] = {
                    'document': r['document'],
                    'dense_score': 0.0,
                    'sparse_score': r['score']
                }

        # 计算加权分数
        for doc in all_docs.values():
            # 归一化
            doc['combined_score'] = (
                self.alpha * self._normalize(doc['dense_score']) +
                (1 - self.alpha) * self._normalize(doc['sparse_score'])
            )

        # 排序返回
        sorted_docs = sorted(
            all_docs.values(),
            key=lambda x: x['combined_score'],
            reverse=True
        )

        return sorted_docs[:top_k]

    def _normalize(self, score: float) -> float:
        # Min-Max 归一化
        return score / (1 + score)
```

### 3.2 数据清洗

```python
import re
from typing import List

class DataCleaner:
    """数据清洗器"""

    def clean(self, text: str) -> str:
        """执行完整的清洗流程"""
        text = self._remove_special_characters(text)
        text = self._normalize_whitespace(text)
        text = self._remove_duplicates(text)
        return text.strip()

    def _remove_special_characters(self, text: str) -> str:
        """移除特殊字符"""
        # 保留中英文、标点和空白
        return re.sub(r'[^\w\s\u4e00-\u9fff,.!?;:()"\']', '', text)

    def _normalize_whitespace(self, text: str) -> str:
        """标准化空白"""
        text = re.sub(r'\s+', ' ', text)  # 多个空白变一个
        text = re.sub(r'\n\s*\n', '\n\n', text)  # 段落分隔
        return text

    def _remove_duplicates(self, text: str) -> str:
        """移除重复内容"""
        lines = text.split('\n')
        seen = set()
        unique_lines = []
        for line in lines:
            if line not in seen:
                seen.add(line)
                unique_lines.append(line)
        return '\n'.join(unique_lines)
```

### 3.3 分块策略 (Chunking Strategies)

#### 3.3.1 固定长度分块

```python
def fixed_size_chunk(text: str, chunk_size: int = 512, overlap: int = 50) -> List[str]:
    """固定长度分块"""
    chunks = []
    start = 0

    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap

    return chunks
```

#### 3.3.3 父子分块 (Parent-Child Chunking)

```python
class ParentChildChunker:
    """父子分块策略"""

    def __init__(self, parent_size: int = 2000, child_size: int = 500):
        self.parent_size = parent_size
        self.child_size = child_size

    def chunk(self, document: str) -> List[Dict]:
        """创建父子分块"""
        parent_chunks = self._create_chunks(document, self.parent_size)
        result = []

        for i, parent in enumerate(parent_chunks):
            # 为每个父分块创建子分块
            child_chunks = self._create_chunks(parent, self.child_size)

            result.append({
                'id': f"parent_{i}",
                'content': parent,
                'type': 'parent',
                'children': [
                    {'id': f"parent_{i}_child_{j}", 'content': child}
                    for j, child in enumerate(child_chunks)
                ]
            })

        return result

    def _create_chunks(self, text: str, size: int) -> List[str]:
        # 简化版分块逻辑
        return [text[i:i+size] for i in range(0, len(text), size - 100)]
```

```python
class MetadataManager:
    """元数据管理器"""

    def __init__(self):
        self.metadata_store = {}

    def add_document(self, doc: Document):
        """添加文档元数据"""
        self.metadata_store[doc.id] = {
            'source': doc.source,
            'created_at': doc.created_at,
            'last_accessed': None,
            'access_count': 0,
            'tags': doc.metadata.get('tags', []),
            'category': doc.metadata.get('category'),
        }

    def filter(self, filters: Dict[str, Any]) -> List[str]:
        """按元数据过滤"""
        results = []
        for doc_id, meta in self.metadata_store.items():
            if all(meta.get(k) == v for k, v in filters.items()):
                results.append(doc_id)
        return results
```

## 4. 检索优化

### 4.1 查询重写 (Query Rewriting)

```python
class QueryRewriter:
    """查询重写器"""

    def __init__(self, llm):
        self.llm = llm

    def rewrite(self, query: str, method: str = 'expansion') -> str:
        """重写查询"""
        if method == 'expansion':
            return self._expand_query(query)
        elif method == 'decompose':
            return self._decompose_query(query)
        elif method == 'hyde':
            return self._hyde(query)
        return query

    def _expand_query(self, query: str) -> str:
        """查询扩展"""
        prompt = f"""Expand the following query with relevant synonyms and related concepts:
Query: {query}
Expanded: """
        return self.llm.generate(prompt)

    def _decompose_query(self, query: str) -> List[str]:
        """查询分解"""
        prompt = f"""Decompose this query into sub-queries:
Query: {query}
Sub-queries: """
        response = self.llm.generate(prompt)
        return [q.strip() for q in response.split('\n') if q.strip()]

    def _hyde(self, query: str) -> str:
        """HyDE - Hypothetical Document Embedding"""
        prompt = f"""Write a hypothetical answer to this query:
Query: {query}
Answer: """
        return self.llm.generate(prompt)
```

### 4.2 重排序 (Re-ranking)

```python
class ReRanker:
    """重排序器"""

    def __init__(self, model_name: str = 'cross-encoder/ms-marco-MiniLM-L-6-v2'):
        from sentence_transformers import CrossEncoder
        self.model = CrossEncoder(model_name)

    def rerank(self, query: str, documents: List[Dict], top_k: int = 5) -> List[Dict]:
        """对检索结果进行重排序"""
        pairs = [[query, doc['content']] for doc in documents]

        # 计算相关性分数
        scores = self.model.predict(pairs)

        # 添加分数到文档
        for doc, score in zip(documents, scores):
            doc['rerank_score'] = float(score)

        # 排序
        ranked = sorted(documents, key=lambda x: x['rerank_score'], reverse=True)
        return ranked[:top_k]
```

```python
import hashlib
from cachetools import TTLCache

class QueryCache:
    """查询缓存"""

    def __init__(self, max_size: int = 10000, ttl: int = 3600):
        self.cache = TTLCache(maxsize=max_size, ttl=ttl)

    def _hash(self, query: str) -> str:
        return hashlib.md5(query.encode()).hexdigest()

    def get(self, query: str) -> Optional[List[Dict]]:
        """从缓存获取结果"""
        key = self._hash(query)
        return self.cache.get(key)

    def set(self, query: str, results: List[Dict]):
        """缓存结果"""
        key = self._hash(query)
        self.cache[key] = results

    def clear(self):
        """清空缓存"""
        self.cache.clear()
```

### 5.2 答案生成

```python
class RAGGenerator:
    """RAG 生成器"""

    def __init__(self, llm, max_tokens: int = 1024):
        self.llm = llm
        self.max_tokens = max_tokens

    def generate(self, query: str, documents: List[Dict]) -> GenerationResult:
        """生成回答"""
        prompt = self._build_prompt(query, documents)

        response = self.llm.generate(
            prompt,
            max_tokens=self.max_tokens,
            temperature=0.1  # 低温度提高事实性
        )

        return GenerationResult(
            answer=response,
            sources=self._extract_sources(response, documents),
            confidence=self._estimate_confidence(response, documents)
        )

    def _estimate_confidence(self, answer: str, documents: List[Dict]) -> float:
        """估计回答可信度"""
        # 基于文档相关性分数和答案长度
        avg_doc_score = sum(d.get('score', 0.5) for d in documents) / len(documents)
        length_factor = min(len(answer) / 100, 1.0)
        return avg_doc_score * 0.7 + length_factor * 0.3
```

### 5.3 引用管理

```python
class CitationManager:
    """引用管理器"""

    def add_citations(self, answer: str, documents: List[Dict]) -> str:
        """添加引用标记"""
        cited_answer = answer

        for i, doc in enumerate(documents):
            citation_marker = f"[{i+1}]"
            # 在相关句子后添加引用
            cited_answer = self._insert_citation(cited_answer, citation_marker, doc)

        # 添加引用列表
        citations = "\n\nReferences:\n"
        for i, doc in enumerate(documents, 1):
            citations += f"[{i}] {doc.get('source', 'Unknown source')}\n"

        return cited_answer + citations

    def _insert_citation(self, text: str, marker: str, doc: Dict) -> str:
        """在合适位置插入引用"""
        # 简单实现：在句末添加
        sentences = text.split('.')
        for i, sentence in enumerate(sentences):
            if any(keyword in sentence.lower() for keyword in doc['content'].lower().split()[:10]):
                sentences[i] = sentence.strip() + marker
                break
        return '.'.join(sentences)
```

## 6. 评估指标与优化方法

### 6.1 评估指标

| 指标 | 描述 | 计算方法 |
|------|------|---------|
| **检索命中率 (Recall@K)** | 相关文档在 top-K 中的比例 | 相关文档数/总相关文档数 |
| **MRR (Mean Reciprocal Rank)** | 首个相关文档排名的倒数平均 | 1/rank 的平均值 |
| **NDCG@K** | 归一化折损累计增益 | 考虑排名位置的加权分数 |
| **答案准确性** | 生成答案的正确性 | 人工评估或 LLM 评判 |
| **引用准确率** | 引用与答案的相关性 | 引用内容支持答案的比例 |
| **响应时间** | 端到端延迟 | 从查询到响应的总时间 |

### 6.2 评估框架

```python
from dataclasses import dataclass
from typing import List

@dataclass
class RAGEvaluation:
    """RAG 评估器"""

    def evaluate(self, queries: List[str], ground_truth: List[str],
                 system_outputs: List[str]) -> EvaluationReport:

        metrics = {
            'answer_correctness': self._measure_correctness(system_outputs, ground_truth),
            'context_relevance': self._measure_context_relevance(queries, system_outputs),
            'faithfulness': self._measure_faithfulness(system_outputs, ground_truth),
            'response_latency': self._measure_latency()
        }

        return EvaluationReport(metrics=metrics)

    def _measure_correctness(self, outputs: List[str], ground_truth: List[str]) -> float:
        """测量答案正确性"""
        # 可以使用 LLM 进行评判
        correct_count = 0
        for output, truth in zip(outputs, ground_truth):
            if self._llm_judge(output, truth):
                correct_count += 1
        return correct_count / len(outputs)

    def _measure_faithfulness(self, outputs: List[str], contexts: List[str]) -> float:
        """测量答案对上下文的忠实度"""
        # 检查答案中的每个陈述是否都能在上下文中找到支持
        faithful_count = 0
        for output, context in zip(outputs, contexts):
            if self._is_faithful(output, context):
                faithful_count += 1
        return faithful_count / len(outputs)
```

### 6.3 优化方法

```python
class RAGOptimizer:
    """RAG 优化器"""

    def __init__(self, retriever, generator, evaluator):
        self.retriever = retriever
        self.generator = generator
        self.evaluator = evaluator

    def optimize_chunk_size(self, documents: List[str],
                            test_queries: List[str]) -> int:
        """优化分块大小"""
        chunk_sizes = [256, 512, 1024, 2048]
        best_size = 512
        best_score = 0

        for size in chunk_sizes:
            # 测试当前分块大小
            chunks = self._chunk_documents(documents, size)
            score = self._evaluate_retrieval(chunks, test_queries)

            if score > best_score:
                best_score = score
                best_size = size

        return best_size

    def optimize_top_k(self, test_queries: List[str]) -> int:
        """优化检索结果数量"""
        k_values = [3, 5, 10, 20]
        best_k = 5
        best_score = 0

        for k in k_values:
            score = self._evaluate_with_k(test_queries, k)
            if score > best_score:
                best_score = score
                best_k = k

        return best_k
```

### 7.1 Advanced RAG

**技术特点：**
- 多跳检索 (Multi-hop Retrieval)
- 自检索 (Self-Retrieval)
- 迭代检索 (Iterative Retrieval)

```python
class AdvancedRAG:
    """Advanced RAG - 迭代检索"""

    def __init__(self, retriever, generator, max_iterations: int = 3):
        self.retriever = retriever
        self.generator = generator
        self.max_iterations = max_iterations

    def query(self, query: str) -> str:
        all_documents = []
        current_query = query

        for i in range(self.max_iterations):
            # 检索
            docs = self.retriever.search(current_query, top_k=5)
            all_documents.extend(docs)

            # 判断是否需要继续检索
            if self._has_sufficient_info(all_documents, query):
                break

            # 生成下一个检索查询
            current_query = self._generate_next_query(query, all_documents)

        return self.generator.generate(query, all_documents)

    def _has_sufficient_info(self, docs: List[Dict], query: str) -> bool:
        # 使用 LLM 判断是否有足够信息
        prompt = f"""Based on these documents, can we answer the query?
Query: {query}
Documents: {[d['content'] for d in docs]}
Answer (yes/no): """
        response = self.generator.llm.generate(prompt)
        return 'yes' in response.lower()
```

### 7.3 Graph RAG

**Graph RAG 特点：**
- 基于知识图谱的检索
- 支持关系推理
- 更好的多跳问答能力

```python
class GraphRAG:
    """Graph-based RAG"""

    def __init__(self, knowledge_graph, retriever, generator):
        self.kg = knowledge_graph
        self.retriever = retriever
        self.generator = generator

    def query(self, query: str) -> str:
        # 提取实体
        entities = self._extract_entities(query)

        # 从知识图谱获取相关子图
        subgraph = self.kg.get_subgraph(entities)

        # 从向量数据库检索文档
        docs = self.retriever.search(query, top_k=5)

        # 结合图谱和文档生成回答
        return self.generator.generate(
            query,
            context={
                'graph': subgraph,
                'documents': docs
            }
        )
```
